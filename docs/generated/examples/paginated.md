# Single-page pagination

Request one page explicitly and inspect metadata before choosing another page.

- Operation: [`GetUniverseGroups`](../operations/GetUniverseGroups.md)
- Repository source: `examples/generated/paginated.ts`

```ts
import { EsiClient } from '@evespace/esi-client';

const client = new EsiClient();

export async function getUniverseGroupPage(page = 1) {
  const response = await client.universe.withMetadata().listItemGroups({ page });
  return {
    groupIds: response.data,
    page,
    pages: response.meta.pagination?.pages,
  };
}
```
